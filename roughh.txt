from main import app
import pytest
import asyncio
# Helper Function for inserting the data directly to database

async def seed_products(products):
    names = [p["name"] for p in products]
    descriptions = [p["description"] for p in products]
    categories = [p["category"] for p in products]
    prices = [p["price"] for p in products]
    stocks = [p["stock"] for p in products]
    ratings = [p["rating"] for p in products]

    async with app.state.pool.acquire() as conn:
        rows = await conn.fetch(
            """
            INSERT INTO products
                (name, description, category, price, stock, rating)
            SELECT *
            FROM UNNEST(
                $1::text[],
                $2::text[],
                $3::text[],
                $4::numeric[],
                $5::integer[],
                $6::numeric[]
            )
            RETURNING id, name, description, category, price, stock, rating
            """,
            names,
            descriptions,
            categories,
            prices,
            stocks,
            ratings
        )

    return [dict(row) for row in rows]

def product_data(**overrides):
    data = {
        "name": "Test Product",
        "description": "Test description",
        "category": "Electronics",
        "price": 1000,
        "stock": 10,
        "rating": 4.5
    }
    data.update(overrides)
    return data

async def get_product_from_db(product_id):
    async with app.state.pool.acquire() as conn:
        row = await conn.fetchrow(
            """
            SELECT id, name, description, category, price, stock, rating
            FROM products
            WHERE id = $1
            """,
            product_id
        )

    return dict(row) if row else None

def patch_client(monkeypatch, outcomes):
    """
    Replace the actual Gemini API call.

    Keep the real Tenacity retry mechanism active.
    """

    calls = {"n": 0}

    async def fake_generate_content(**kwargs):
        outcome = outcomes[calls["n"]]
        calls["n"] += 1

        if isinstance(outcome, Exception):
            raise outcome

        return outcome

    monkeypatch.setattr(
        "gemini_helper.client.aio.models.generate_content",
        fake_generate_content
    )

    monkeypatch.setattr(
        "gemini_helper._call_with_retry",
        gh._call_with_retry.retry_with(wait=wait_none())
    )
    return calls

# Test-8 
async def test_failed_product_cancels_remaining_tasks(client, monkeypatch):

    a_started = asyncio.Event()
    a_cancelled = asyncio.Event()

    async def fake_process_product(product):

        if product.name == "Product A":

            a_started.set()

            try:
                # Keep Product A running until it gets cancelled.
                await asyncio.Future()

            except asyncio.CancelledError:
                a_cancelled.set()
                raise

        elif product.name == "Product B":

            # Wait until Product A has actually started.
            await a_started.wait()

            # Simulate a processing failure.
            raise RuntimeError("Simulated processing failure")

    monkeypatch.setattr(
        "main.process_product",
        fake_process_product
    )

    response = await client.post(
        "/products",
        json=[
            {
                "name": "Product A",
                "description": "Description for product A",
                "category": "Electronics",
                "price": 1000,
                "stock": 5,
                "rating": 4.0,
                "ai_description": False
            },
            {
                "name": "Product B",
                "description": "Description for product B",
                "category": "Electronics",
                "price": 1500,
                "stock": 8,
                "rating": 4.5,
                "ai_description": False
            }
        ]
    )

    assert response.status_code == 500
    assert a_cancelled.is_set()

# Test-13

async def test_bulk_ai_success(client, monkeypatch):

    async def fake_call_gemini(prompt):
        if "Product A" in prompt:
            return "Description for Product A."

        if "Product B" in prompt:
            return "Description for Product B."

        raise AssertionError("Unexpected product in prompt")

    monkeypatch.setattr("main.call_gemini", fake_call_gemini)

    response = await client.post(
        "/products",
        json=[
            {
                "name": "Product A",
                "price": 100,
                "stock": 5,
                "ai_description": True
            },
            {
                "name": "Product B",
                "price": 200,
                "stock": 10,
                "ai_description": True
            }
        ]
    )

    assert response.status_code == 200

    data = response.json()

    assert len(data) == 2

    assert data[0]["description"] == "Description for Product A."
    assert data[1]["description"] == "Description for Product B."
    async with app.state.pool.acquire() as conn:
        rows = await conn.fetch(
            """
            SELECT name, description
            FROM products
            WHERE name = ANY($1::text[])
            """,
            ["Product A", "Product B"]
        )

    saved_products = {
        row["name"]: row["description"]
        for row in rows
    }

    assert len(saved_products) == 2

    assert saved_products == {
        "Product A": "Description for Product A.",
        "Product B": "Description for Product B."
    }
